## 第一步 本地Node服务
<font color=Red>注意</font>
由于浏览器跨域问题，在开发中需要使用nodejs给发送的连接做一次转发，若没有跨域问题，可以跳过此步骤

下载完demo后我们先进入server目录，安装依赖后起node服务
```nodejs
cd server
npm install
npm run server
```

关于请求接口鉴权可以参考下列示例，详细参数解释请看 [鉴权-公共参数](https://open-live.bilibili.com/document/doc&tool/auth.html#%E5%85%AC%E5%85%B1%E5%8F%82%E6%95%B0)
```typescript
/**
 * 鉴权加密
 * @param {*} params
 * @returns
 */
export function getEncodeHeader(
    params = {},
    appKey: string,
    appSecret: string
) {
    const timestamp = parseInt(Date.now()+ "")
    const nonce = parseInt(Math.random() * 100000000 + "") + timestamp
    const header = {
        "x-bili-accesskeyid": appKey,
        // 请求体的编码值，根据请求体计算所得。算法说明：将请求体内容当作字符串进行MD5编码。
        "x-bili-content-md5": getMd5Content(JSON.stringify(params)),
        // 签名方式。取值：HMAC-SHA256
        "x-bili-signature-method": "HMAC-SHA256",
        // 签名唯一随机数。用于防止网络重放攻击，建议您每一次请求都使用不同的随机数。
        "x-bili-signature-nonce": nonce + "",
        "x-bili-signature-version": "1.0",
        // unix时间戳，单位是秒。请求时间戳不能超过当前时间10分钟，否则请求会被丢弃。
        "x-bili-timestamp": timestamp
    }
    const data: string[] = []
    for (const key in header) {
        data.push(`${key}:${header[key]}`)
    }

    const signature = crypto
        .createHmac("sha256", appSecret)
        .update(data.join("\n"))
        .digest("hex")
    return {
        // 接受的返回结果的类型。目前只支持JSON类型，取值：application/json。
        Accept: "application/json",
        "Content-Type": "application/json",
        ...header,
        Authorization: signature
    }
}

/**
 * MD5加密
 * @param {*} str
 * @returns
 */
export function getMd5Content(str) {
    return crypto
        .createHash("md5")
        .update(str)
        .digest("hex")
}

```

## 第二步 客户端使用

客户端部分进入client目录，此处我们使用vue代码做示例
```nodejs
cd client
npm install
npm run dev
```
<font color=Red>注意</font>
appKey和appSecret在demo中写在客户端传入接口是便于调试，出于安全考虑，可以写入server中


接口在node服务进行了鉴权以及转发解决跨域后，直接调用接口即可跑通接口，以gameStart服务为例
```typescript
/**
 * @comment 注意所有的接口基于鉴权成功后才能正确返回
 * 测试请求游戏开启接口 v2版本
 */
const gameStart = () => {
    api.post("/gameStart", {
        code: codeId.value,
        app_id: Number(appId.value),
    })
        .then(({ data }) => {
            if (data.code === 0) {
                const res = data.data
                const { game_info, websocket_info } = res
                const { auth_body, wss_link } = websocket_info
                authBody.value = auth_body
                wssLinks.value = wss_link
                console.log("-----游戏开始成功-----")
                console.log("返回GameId：", game_info)
                gameId.value = game_info.game_id
                // v2改为20s请求心跳一次，不然60s会自动关闭
                heartBeatTimer.value = setInterval(() => {
                    heartBeatThis(game_info.game_id)
                }, 20000)
            } else {
                console.log("-----游戏开始失败-----")
                console.log("原因：", data)
            }
        })
        .catch((err) => {
            console.log("-----游戏开始失败-----")
            console.log(err)
        })
}

```
